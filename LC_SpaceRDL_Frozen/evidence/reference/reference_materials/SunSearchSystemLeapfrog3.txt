Import DeviceRepository;
Import CtrComputerRepository; 
Import PortRepository;
Import DataDictionary;

System{
Device Gyro3F gyro1;
Device DSS dss1;
Device JET jet1;
Device ExtComputer obdh1;
Controller AOCC aocc{ ControlPeriod: 160ms};
Connect UARTport1 gyro1; /*将aocc的UARTport1与gyro1连接*/
Connect {PULSEport3, ADport5, ADport6, ADport7} dss1; /*将aocc的PULSEport3, ADport5,  ADport6, ADport7与dss1连接*/
Connect { PULSEport4,ADport6 } jet1; /*将aocc的PULSEport4,ADport6与jet1连接*/
Connect UARTport2 obdh1; /*将aocc的UARTport2与obdh1连接*/
}
InitTask{ /*初始化任务*/
Init UARTport1 {1,1,115200};/*初始化与陀螺通信端口UARTport1,设置停止位、校验位和波特率为1,1,115200 ,不同接口具体初始化模板见portRepository*/
Init UARTport2 {1,1,38400};/*初始化与obdh通信端口UARTport2,设置停止位、校验位和波特率为1,1,38400 */
SendData dss1 DssPowerOn;/*将数据powerOn发送给设备dss1,在设备库中找到设备dss1中poweron数据对应的端口和协议*/
SendData gyro1 GyroPowerOn;/*将数据powerOn发送给设备gyro1,在设备库中找到设备gyro1中poweron数据对应的端口和协议*/
SendData jet1 JetPowerOn;/*将数据powerOn发送给设备jet1,在设备库中找到设备jet1中poweron数据对应的端口和协议*/
Init timer32 1;    /* 初始化定时器,向定时器寄存器地址中写入1,在控制计算机库中找到定时器timer32的寄存器地址,启动定时器,寄存器配置说明见CtrComputerRepository */
}

AppTask{
Set cntSate  cntSate + 1;
Set cntMode  cntMode + 1;
ProTC obdh1 {recvFlag,workmode} ReqCapByTable ProTable.TC;/*从obdh1中获得遥控数据：通过UARTport2,按照协议表ProTable中的TC条目的格式获得工作模式,并根据包头、长度和校验和的判断设置recvFlag */
If(recvFlag==true){	
Set nextmode workmode;}
GetData gyro1 {deltG[9],GyroPowerState[9]};/* 9陀螺数据采集：通过UARTport1按照gyro_proTable1格式发送取数指令,等待大于其响应时间后,按照gyro_proTable3格式获得陀螺角速度和加电状态*/    
GetData dss1 {viewCode,AttigDss[2],DssPowerState[2]};/*2太阳敏感器数据采集：通过ADport5和ADport6,获得太阳敏感器的可见标志和加电状态,通过ADport7,按照12位偏移二进制到浮点数的转换获太阳方位角*/
GetData jet1 {JetPowerState[12]};/*12推力器数据采集：通过ADport6,获得推力器加电状态*/ 
ProVld { deltG[9]} {deltG[9]} ReqCapByTable FormulaTable.B1;/*陀螺数据剔野：根据公式表B1条目进行陀螺角速度的剔野处理*/
DetAtt {wG,attiangG} {attiangG,deltG[9] ,GyroPowerState[9]} ReqCapByTable FormulaTable.B2;/*陀螺姿态确定：根据公式表B2条目,通过陀螺数据计算得到三轴陀螺惯性角速度和积分角度*/
DetAtt {attiang,attirate,attiangG } { currentmode,attiangG,wG,viewCode,AttigDss} ReqCapByTable FormulaTable.B3;/*陀螺与太阳敏感器姿态确定：根据公式表B3条目,利用模式字、陀螺计算得到的姿态角和太阳方位角,确定用于控制的三轴姿态角和角速度 */
Composite modeProcess{/*模式处理 */ 
SwitchMode {nextmode} {currentmode,viewCode, cntMode,attirate} ReqCapByTable FormulaTable.B5;/*模式切换 ：根据根据公式表B5条目,进行模式切换条件的判断和处理*/
If (currentmode!= nextmode){
Set currentmode nextmode;/*模式初始化 */
Set cntMode 0; }
Switch(nextmode):{
Case(RDSM):{ Set attirateT {0.0,0.0,0.0};/* RDSM目标姿态计算：设置三轴目标角速度为{0.0,0.0,0.0} */}
Case(PASM): { Set attirateT {0.0,0.5,0.0};/* PASM目标姿态计算：设置三轴目标角速度为{0.0,0.5,0.0}*/}
Case(RASM): { Set attirateT {0.5,0.0,0.0};/* RASM目标姿态计算：设置三轴目标角速度为{0.5,0.0,0.0}*/}
Case(CSM): { Set attirateT {0.0,0.0,0.0};  /* CSM目标姿态计算：设置三轴目标角速度为{0.0,0.0,0.0} */}
}}
CalCtrl pulseWidth {attiang,attirate,attiangT,attirateT} ReqCapByTable FormulaTable.B4;/*推力器数据控制计算：根据公式表B4条目进行计算,此功能包括三个顺序的处理过程*/
ProTM obdh1 {workmode,attiang,attirate,cntSate} ReqCapByTable ProTable.TM;/*遥测处理：通过UARTport2按照proTable表TM条目格式进行遥测数据的打包发送 */
}Finished Within 130 ms  /*在130ms内完成 */

InterruptTask{
At(128ms ,Period)  SendData gyro1 pulseWidth Finished Within 100us;/*在每个周期的128ms,通过PULSEport4,发送推力器脉宽指令,中断处理时间要求在100us内完成 */
}
/*对于整体系统的时间、空间、安全可靠的约束 */
SystemConstraints{
Executed Within 32 kb; /*空间约束为32kb */
}
